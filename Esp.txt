#include <vector>
#include <cmath>
#include <algorithm>
#include <cstdint>

struct Vec3 { float x, y, z; };
struct Vec2 { float x, y; };
struct Color { uint8_t r, g, b, a; };

struct Matrix4 {
    float m[4][4];
    bool WorldToScreen(const Vec3& p, int sw, int sh, Vec2& out) const {
        float w = m[3][0] * p.x + m[3][1] * p.y + m[3][2] * p.z + m[3][3];
        if (w < 0.001f) return false;
        float inv = 1.0f / w;
        float x = (m[0][0] * p.x + m[0][1] * p.y + m[0][2] * p.z + m[0][3]) * inv;
        float y = (m[1][0] * p.x + m[1][1] * p.y + m[1][2] * p.z + m[1][3]) * inv;
        out.x = (sw * 0.5f) * (1.0f + x);
        out.y = (sh * 0.5f) * (1.0f - y);
        return true;
    }
};

struct Entity {
    int id;
    int team;
    int health;
    bool alive;
    Vec3 origin;
    float height;
    const char* name;
};

class IWorld {
public:
    virtual ~IWorld() = default;
    virtual const std::vector<Entity*>& Entities() const = 0;
    virtual Entity* LocalPlayer() const = 0;
    virtual Matrix4 ViewMatrix() const = 0;
    virtual void ScreenSize(int& w, int& h) const = 0;
    virtual bool HasLineOfSight(const Vec3& from, const Vec3& to) const = 0;
};

class IRenderer {
public:
    virtual ~IRenderer() = default;
    virtual void Line(Vec2 a, Vec2 b, Color c, float thickness) = 0;
    virtual void Rect(Vec2 min, Vec2 max, Color c, float thickness) = 0;
    virtual void FilledRect(Vec2 min, Vec2 max, Color c) = 0;
    virtual void Text(Vec2 pos, const char* text, Color c, bool centered) = 0;
};

struct ESPConfig {
    bool enabled = true;
    bool box = true;
    bool healthBar = true;
    bool nameTag = true;
    bool distance = true;
    bool snapLines = false;
    bool teamCheck = true;
    bool visibleOnlyColor = true;
    float maxDistance = 5000.0f;
    Color enemyVisible { 255, 60, 60, 255 };
    Color enemyHidden { 255, 190, 0, 255 };
    Color ally { 80, 160, 255, 255 };
};

class ESPSystem {
public:
    explicit ESPSystem(IWorld* w, IRenderer* r) : world(w), renderer(r) {}

    ESPConfig& Config() { return cfg; }

    void Render() {
        if (!cfg.enabled) return;
        Entity* me = world->LocalPlayer();
        if (!me || !me->alive) return;

        int sw, sh;
        world->ScreenSize(sw, sh);
        Matrix4 vm = world->ViewMatrix();

        for (Entity* e : world->Entities()) {
            if (!e || e == me || !e->alive) continue;
            bool isAlly = e->team == me->team;
            if (cfg.teamCheck && isAlly) continue;

            float dist = Distance(me->origin, e->origin);
            if (dist > cfg.maxDistance) continue;

            Vec3 head { e->origin.x, e->origin.y, e->origin.z + e->height };
            Vec2 sFeet, sHead;
            if (!vm.WorldToScreen(e->origin, sw, sh, sFeet)) continue;
            if (!vm.WorldToScreen(head, sw, sh, sHead)) continue;

            Color col = PickColor(me, e, isAlly);
            float boxH = sFeet.y - sHead.y;
            float boxW = boxH * 0.45f;
            Vec2 tl { sHead.x - boxW * 0.5f, sHead.y };
            Vec2 br { sHead.x + boxW * 0.5f, sFeet.y };

            if (cfg.box) renderer->Rect(tl, br, col, 1.5f);
            if (cfg.healthBar) DrawHealth(tl, br, e->health);
            if (cfg.nameTag) renderer->Text({ sHead.x, tl.y - 14.0f }, e->name, col, true);
            if (cfg.distance) {
                char buf[32];
                snprintf(buf, sizeof(buf), "%dm", (int)(dist / 39.37f));
                renderer->Text({ sHead.x, br.y + 2.0f }, buf, col, true);
            }
            if (cfg.snapLines) renderer->Line({ sw * 0.5f, (float)sh }, { sHead.x, br.y }, col, 1.0f);
        }
    }

private:
    IWorld* world;
    IRenderer* renderer;
    ESPConfig cfg;

    static float Distance(const Vec3& a, const Vec3& b) {
        float dx = a.x - b.x, dy = a.y - b.y, dz = a.z - b.z;
        return std::sqrt(dx * dx + dy * dy + dz * dz);
    }

    Color PickColor(Entity* me, Entity* e, bool isAlly) {
        if (isAlly) return cfg.ally;
        if (!cfg.visibleOnlyColor) return cfg.enemyVisible;
        Vec3 eye { me->origin.x, me->origin.y, me->origin.z + me->height };
        Vec3 target { e->origin.x, e->origin.y, e->origin.z + e->height * 0.9f };
        return world->HasLineOfSight(eye, target) ? cfg.enemyVisible : cfg.enemyHidden;
    }

    void DrawHealth(Vec2 tl, Vec2 br, int hp) {
        float pct = std::clamp(hp / 100.0f, 0.0f, 1.0f);
        float h = br.y - tl.y;
        Vec2 barMin { tl.x - 6.0f, tl.y };
        Vec2 barMax { tl.x - 3.0f, br.y };
        renderer->FilledRect(barMin, barMax, { 0, 0, 0, 160 });
        Color c { (uint8_t)(255 * (1.0f - pct)), (uint8_t)(255 * pct), 0, 255 };
        renderer->FilledRect({ barMin.x, br.y - h * pct }, barMax, c);
    }
};
